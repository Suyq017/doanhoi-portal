\textbf{Actors} 
& Quản trị viên \\ \hline

\textbf{Description} 
& Use case này cho phép quản trị viên thực hiện các thao tác quản lý thông tin đơn vị trong hệ thống bao gồm:
\begin{itemize}
    \item Xem danh sách đơn vị
    \item Tìm và lọc đơn vị
    \item Thực hiện các thao tác thêm, xóa, sửa thông tin đơn vị
\end{itemize}
\\ \hline

\textbf{Trigger} 
& Quản trị viên truy cập chức năng quản lý đơn vị từ giao diện hệ thống. 
\\ \hline

\textbf{Pre-Conditions} 
& Quản trị viên đã đăng nhập vào hệ thống bằng tài khoản có phân quyền quản lý đơn vị. 
\\ \hline

\textbf{Post-Conditions} 
& Thông tin đơn vị được hiển thị và cập nhật đúng theo thao tác đã thực hiện. 
\\ \hline

\textbf{Normal Flow} 
& 
\begin{enumerate}
    \item Quản trị viên truy cập module \textit{Quản lý đơn vị}.
    \item Hệ thống hiển thị danh sách đơn vị.
    \item Quản trị viên chọn một trong các thao tác:
    \begin{itemize}
        \item Xem danh sách đơn vị
        \item Tìm và lọc đơn vị
        \item Thực hiện các thao tác thêm, xóa, sửa thông tin đơn vị
    \end{itemize}
\end{enumerate}
\\ \hline

\textbf{Alternative Flow} 
& Ở bước 3:
\begin{itemize}
    \item \textbf{3a.} Quản trị viên thực hiện tìm kiếm/lọc đơn vị, hệ thống hiển thị danh sách đơn vị phù hợp với điều kiện tìm kiếm.
\end{itemize}
\\ \hline

\textbf{Exception} 
& Ở bước 1:
\begin{itemize}
    \item \textbf{1a.} Nếu quản trị viên không có phân quyền truy cập vào chức năng quản lý đơn vị thì hệ thống sẽ chặn truy cập và không hiển thị module này ở phía giao diện người dùng.
\end{itemize}
